/* ========================================================
   ANIMATION 2: Orientation Canvases (Slide 2)
   ======================================================== */
(function() {
    const cvsCorrect = document.getElementById('orientation-correct-canvas');
    const cvsIncorrect = document.getElementById('orientation-incorrect-canvas');
    if (!cvsCorrect || !cvsIncorrect) return;
    
    const c1 = cvsCorrect.getContext('2d');
    const c2 = cvsIncorrect.getContext('2d');
    
    let animId;
    let progress = 0; // 0 to 1
    
    function drawAtom(ctx, x, y, r, color1, color2, label, labelColor = '#fff') {
        const grd = ctx.createRadialGradient(x - r*0.3, y - r*0.3, r*0.1, x, y, r);
        grd.addColorStop(0, color1);
        grd.addColorStop(1, color2);
        ctx.beginPath();
        ctx.arc(x, y, r, 0, Math.PI*2);
        ctx.fillStyle = grd;
        ctx.fill();
        
        ctx.font = `bold ${Math.floor(r)}px sans-serif`;
        ctx.fillStyle = labelColor;
        ctx.textAlign = 'center';
        ctx.textBaseline = 'middle';
        ctx.fillText(label, x, y);
    }
    
    function drawCorrect() {
        const W = cvsCorrect.width, H = cvsCorrect.height;
        c1.clearRect(0, 0, W, H);
        c1.fillStyle = 'rgba(15,23,42,0.95)';
        c1.fillRect(0, 0, W, H);
        
        const cx = W / 2, cy = H / 2;
        
        // I- (attacking from left)
        // Transition from x = 40 to cx - 40
        // Cl- (leaving to right)
        // Transition from cx + 40 to W - 40
        
        let iX, clX, hBend; 
        // hBend: -1 (left bent), 0 (planar), 1 (right bent)
        
        if (progress < 0.4) {
            // Approaching
            let p = progress / 0.4; // 0 to 1
            iX = 40 + p * (cx - 70 - 40);
            clX = cx + 45;
            hBend = -1;
        } else if (progress < 0.6) {
            // Transition state
            let p = (progress - 0.4) / 0.2; // 0 to 1
            iX = cx - 70 + p * 15;
            clX = cx + 45 + p * 15;
            hBend = -1 + p * 2; // -1 to 1
        } else {
            // Leaving
            let p = (progress - 0.6) / 0.4; // 0 to 1
            iX = cx - 55 + p * 10;
            clX = cx + 60 + p * (W - 40 - (cx + 60));
            hBend = 1;
        }
        
        // Draw I
        drawAtom(c1, iX, cy, 22, '#c084fc', '#7e22ce', 'I⁻');
        // Draw Cl
        drawAtom(c1, clX, cy, 20, '#6ee7b7', '#047857', progress >= 0.7 ? 'Cl⁻' : 'Cl');
        
        // Draw C (center)
        const cX = cx;
        
        // Bonds to H (depending on hBend)
        // H1 (up), H2 (down-back), H3 (down-front)
        const hDist = 35;
        const bendAng = hBend * 0.5; // rad
        
        c1.strokeStyle = 'rgba(255,255,255,0.4)';
        c1.lineWidth = 3;
        
        // H positions
        const hy1 = cy - hDist * Math.cos(bendAng);
        const hx1 = cX + hDist * Math.sin(bendAng);
        
        const hy2 = cy + hDist * 0.5 * Math.cos(bendAng);
        const hx2 = cX + hDist * Math.sin(bendAng);
        
        // Draw bonds
        c1.beginPath(); c1.moveTo(cX, cy); c1.lineTo(hx1, hy1); c1.stroke();
        c1.beginPath(); c1.moveTo(cX, cy); c1.lineTo(hx2, hy2 - 15); c1.stroke();
        c1.beginPath(); c1.moveTo(cX, cy); c1.lineTo(hx2, hy2 + 15); c1.stroke();
        
        // Draw partially forming/breaking bonds in TS
        if (progress > 0.3 && progress < 0.7) {
            c1.strokeStyle = 'rgba(255,255,255,0.2)';
            c1.setLineDash([4, 4]);
            c1.beginPath(); c1.moveTo(iX + 22, cy); c1.lineTo(cX - 18, cy); c1.stroke();
            c1.beginPath(); c1.moveTo(cX + 18, cy); c1.lineTo(clX - 20, cy); c1.stroke();
            c1.setLineDash([]);
        } else if (progress >= 0.7) {
            // New I-C bond
            c1.strokeStyle = 'rgba(255,255,255,0.4)';
            c1.beginPath(); c1.moveTo(iX + 22, cy); c1.lineTo(cX - 18, cy); c1.stroke();
        } else if (progress <= 0.3) {
            // Old C-Cl bond
            c1.strokeStyle = 'rgba(255,255,255,0.4)';
            c1.beginPath(); c1.moveTo(cX + 18, cy); c1.lineTo(clX - 20, cy); c1.stroke();
        }
        
        // Draw C
        drawAtom(c1, cX, cy, 18, '#94a3b8', '#334155', 'C');
        
        // Draw H
        drawAtom(c1, hx1, hy1, 10, '#f1f5f9', '#94a3b8', 'H', '#0f172a');
        drawAtom(c1, hx2, hy2 - 15, 10, '#f1f5f9', '#94a3b8', 'H', '#0f172a');
        drawAtom(c1, hx2, hy2 + 15, 10, '#f1f5f9', '#94a3b8', 'H', '#0f172a');
    }
    
    function drawIncorrect() {
        const W = cvsIncorrect.width, H = cvsIncorrect.height;
        c2.clearRect(0, 0, W, H);
        c2.fillStyle = 'rgba(15,23,42,0.95)';
        c2.fillRect(0, 0, W, H);
        
        const cx1 = W * 0.25, cy = H / 2 + 15;
        const cx2 = W * 0.75;
        
        function drawScenario(cx, cy, attackFromLeft) {
            // Draw CH3Cl
            const cX = cx - 15;
            const clX = cX + 45;
            
            // C-Cl bond
            c2.strokeStyle = 'rgba(255,255,255,0.4)'; c2.lineWidth = 3;
            c2.beginPath(); c2.moveTo(cX, cy); c2.lineTo(clX, cy); c2.stroke();
            
            // H bonds
            const hDist = 35;
            const bendAng = -0.5; // Up-left
            const hy1 = cy - hDist * Math.cos(bendAng);
            const hx1 = cX + hDist * Math.sin(bendAng);
            
            const hy2 = cy + hDist * 0.5 * Math.cos(bendAng);
            const hx2 = cX + hDist * Math.sin(bendAng);
            
            c2.beginPath(); c2.moveTo(cX, cy); c2.lineTo(hx1, hy1); c2.stroke();
            c2.beginPath(); c2.moveTo(cX, cy); c2.lineTo(hx2, hy2 - 15); c2.stroke();
            c2.beginPath(); c2.moveTo(cX, cy); c2.lineTo(hx2, hy2 + 15); c2.stroke();
            
            drawAtom(c2, cX, cy, 18, '#94a3b8', '#334155', 'C'); // Grey
            drawAtom(c2, hx1, hy1, 10, '#f8fafc', '#94a3b8', 'H', '#0f172a'); // White
            drawAtom(c2, hx2, hy2 - 15, 10, '#f8fafc', '#94a3b8', 'H', '#0f172a');
            drawAtom(c2, hx2, hy2 + 15, 10, '#f8fafc', '#94a3b8', 'H', '#0f172a');
            drawAtom(c2, clX, cy, 22, '#6ee7b7', '#047857', 'Cl', '#0f172a'); // Green
            
            if (attackFromLeft) {
                // I- attacks H from top-left, aligned with C-H bond
                const dx = hx1 - cX;
                const dy = hy1 - cy;
                const len = Math.hypot(dx, dy);
                const nx = dx / len;
                const ny = dy / len;
                
                const dist = 60;
                const iX = hx1 + nx * dist;
                const iY = hy1 + ny * dist;
                
                drawAtom(c2, iX, iY, 26, '#c084fc', '#7e22ce', 'I⁻', '#fff'); // Purple
                
                // Motion lines (parallel to the movement)
                c2.strokeStyle = 'rgba(192,132,252,0.5)';
                c2.lineWidth = 1.5;
                for(let i=-2; i<=2; i++) {
                    c2.beginPath();
                    const px = iX + ny * (i * 6); // perpendicular offset
                    const py = iY - nx * (i * 6);
                    c2.moveTo(px + nx * 18, py + ny * 18);
                    c2.lineTo(px + nx * 50, py + ny * 50);
                    c2.stroke();
                }
                
                // Red Arrow
                c2.strokeStyle = '#ef4444'; c2.lineWidth = 2.5;
                const arrStartX = iX - nx * 30;
                const arrStartY = iY - ny * 30;
                const arrEndX = hx1 + nx * 14;
                const arrEndY = hy1 + ny * 14;
                
                c2.beginPath(); c2.moveTo(arrStartX, arrStartY); c2.lineTo(arrEndX, arrEndY); c2.stroke();
                
                // Arrowhead
                const angle = Math.atan2(-ny, -nx);
                c2.beginPath(); 
                c2.moveTo(arrEndX, arrEndY); 
                c2.lineTo(arrEndX - 10 * Math.cos(angle - Math.PI/6), arrEndY - 10 * Math.sin(angle - Math.PI/6)); 
                c2.lineTo(arrEndX - 10 * Math.cos(angle + Math.PI/6), arrEndY - 10 * Math.sin(angle + Math.PI/6)); 
                c2.fillStyle = '#ef4444'; c2.fill();
            } else {
                // I- attacks Cl from top-right. Let's make an angle.
                const nx = Math.cos(-Math.PI/4); // top-right
                const ny = Math.sin(-Math.PI/4);
                
                const dist = 60;
                const iX = clX + nx * dist;
                const iY = cy + ny * dist;
                
                drawAtom(c2, iX, iY, 26, '#c084fc', '#7e22ce', 'I⁻', '#fff'); // Purple
                
                // Motion lines
                c2.strokeStyle = 'rgba(192,132,252,0.5)';
                c2.lineWidth = 1.5;
                for(let i=-2; i<=2; i++) {
                    c2.beginPath();
                    const px = iX + ny * (i * 6); 
                    const py = iY - nx * (i * 6);
                    c2.moveTo(px + nx * 18, py + ny * 18);
                    c2.lineTo(px + nx * 50, py + ny * 50);
                    c2.stroke();
                }
                
                // Red Arrow
                c2.strokeStyle = '#ef4444'; c2.lineWidth = 2.5;
                const arrStartX = iX - nx * 30;
                const arrStartY = iY - ny * 30;
                const arrEndX = clX + nx * 26;
                const arrEndY = cy + ny * 26;
                
                c2.beginPath(); c2.moveTo(arrStartX, arrStartY); c2.lineTo(arrEndX, arrEndY); c2.stroke();
                
                // Arrowhead
                const pointAngle = Math.atan2(-ny, -nx); 
                c2.beginPath(); 
                c2.moveTo(arrEndX, arrEndY); 
                c2.lineTo(arrEndX - 10 * Math.cos(pointAngle - Math.PI/6), arrEndY - 10 * Math.sin(pointAngle - Math.PI/6)); 
                c2.lineTo(arrEndX - 10 * Math.cos(pointAngle + Math.PI/6), arrEndY - 10 * Math.sin(pointAngle + Math.PI/6)); 
                c2.fillStyle = '#ef4444'; c2.fill();
            }
        }
        
        drawScenario(cx1, cy, true);
        drawScenario(cx2, cy, false);
    }
    
    function animate() {
        if (!document.getElementById('slide-orientation').classList.contains('active')) {
            animId = requestAnimationFrame(animate); return;
        }
        progress += 0.005;
        if (progress > 1.2) progress = 0; // Loop with some pause
        
        drawCorrect();
        drawIncorrect(); // Static but redraws when active
        
        animId = requestAnimationFrame(animate);
    }
    
    function onResize() {
        cvsCorrect.width = cvsCorrect.offsetWidth;
        cvsCorrect.height = cvsCorrect.offsetHeight;
        cvsIncorrect.width = cvsIncorrect.offsetWidth;
        cvsIncorrect.height = cvsIncorrect.offsetHeight;
        drawIncorrect();
    }
    
    window.addEventListener('resize', onResize);
    setTimeout(() => { onResize(); animate(); }, 100);
})();

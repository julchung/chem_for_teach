function showMechCase(n) {
    currentMechCase = n;
    [1,2,3].forEach(i => {
        document.getElementById('mech-case-' + i).style.display = (i === n) ? 'flex' : 'none';
        const btn = document.getElementById('mech-btn-' + i);
        btn.classList.toggle('mech-tab-active', i === n);
    });
    drawMechPracticeCanvas(n);
    if (window.MathJax && window.MathJax.typesetPromise) {
        MathJax.typesetPromise([document.getElementById('slide-mechanism-practice')]).catch(()=>{});
    }
}

    }
}
